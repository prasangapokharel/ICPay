### 1. Minimal Launch Form (`SunPumpLauncher.tsx`)

This component handles image selection, metadata inputs, optional socials, and initial ICP buy amount. It executes transfers via your wallet context (`useWallet` / `authActor`).

```tsx
"use client";

import React, { useState } from "react";

export const USERNAME_TREASURY = "ni5n2-efxui-dyqdu-2mnpr-atclq-d6snc-zdq5q-u6ibz-ibpkq-brjpj-gqe";
export const BASE_LAUNCH_FEE_ICP = 5;

export default function SunPumpLauncher() {
  const [image, setImage] = useState<File | null>(null);
  const [imagePreview, setImagePreview] = useState<string | null>(null);
  const [name, setName] = useState("");
  const [symbol, setSymbol] = useState("");
  const [description, setDescription] = useState("");
  
  // Social Links
  const [website, setWebsite] = useState("");
  const [telegram, setTelegram] = useState("");
  const [twitter, setTwitter] = useState("");
  
  // Initial Buy & Pair Toggle
  const [initialBuyIcp, setInitialBuyIcp] = useState("0");
  const [loading, setLoading] = useState(false);

  // Dynamic Total Calculation (5 ICP Base Fee + Initial Buy/Liquidity Amount)
  const initialBuyNum = parseFloat(initialBuyIcp || "0");
  const totalDeductionIcp = BASE_LAUNCH_FEE_ICP + initialBuyNum;

  const handleImageChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (e.target.files && e.target.files[0]) {
      const file = e.target.files[0];
      setImage(file);
      setImagePreview(URL.createObjectURL(file));
    }
  };

  const handleLaunch = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);

    try {
      const totalE8s = BigInt(Math.round(totalDeductionIcp * 100_000_000));
      const initialBuyE8s = BigInt(Math.round(initialBuyNum * 100_000_000));

      // 1. Direct Transfer to Treasury using your internal wallet actor (No Plug)
      // Assuming `walletActor.transfer` is your custom identity caller
      const txResult = await window.walletActor.icrc1_transfer({
        to: { owner: USERNAME_TREASURY, subaccount: [] },
        amount: totalE8s,
      });

      if (!("Ok" in txResult)) {
        throw new Error("ICP payment to treasury failed.");
      }

      // 2. Upload image to outer CDN/R2 or directly pass base64
      let imageUrl = "";
      if (image) {
        // Simple base64 or upload service
        imageUrl = await uploadToCDN(image); 
      }

      // 3. Call Motoko Factory Canister to deploy child ICRC-1 ledger
      const deployResult = await window.factoryActor.deployToken({
        name,
        symbol: symbol.toUpperCase(),
        description,
        imageUrl,
        website,
        telegram,
        twitter,
        initialBuyE8s,
        txBlockHeight: txResult.Ok, // Block index from transaction
      });

      if (deployResult.success) {
        alert(`🚀 Token Launched Successfully! Canister ID: ${deployResult.canisterId}`);
      }
    } catch (err: any) {
      alert(err.message || "Launch process failed");
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-lg mx-auto p-6 bg-[#0c0d10] border border-slate-800 rounded-2xl text-white font-sans">
      {/* Banner */}
      <div className="flex gap-4 mb-6 text-xs text-slate-400 border-b border-slate-800 pb-4">
        <span>⚡ No Presale</span>
        <span>🛡️ No Team Allocation</span>
        <span>🔥 Lower Gas</span>
      </div>

      <form onSubmit={handleLaunch} className="space-y-4">
        {/* Image & Main Metadata */}
        <div className="flex gap-4">
          <label className="w-32 h-32 border-2 border-dashed border-slate-800 rounded-xl flex flex-col items-center justify-center cursor-pointer hover:border-amber-400 bg-slate-950 overflow-hidden relative">
            {imagePreview ? (
              <img src={imagePreview} alt="Token preview" className="w-full h-full object-cover" />
            ) : (
              <div className="text-center p-2 text-slate-500 text-[10px]">
                <span>JPEG/PNG/WEBP</span>
                <span className="block text-[9px]">Less Than 4MB</span>
              </div>
            )}
            <input type="file" accept="image/*" onChange={handleImageChange} className="hidden" />
          </label>

          <div className="flex-1 space-y-3">
            <div>
              <label className="text-[10px] font-medium text-slate-400">Token Name *</label>
              <input
                type="text"
                placeholder="e.g. Sun Pump ICP"
                value={name}
                onChange={(e) => setName(e.target.value)}
                required
                className="w-full mt-1 bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-xs focus:outline-none focus:border-amber-400"
              />
            </div>
            <div>
              <label className="text-[10px] font-medium text-slate-400">Token Symbol ($Ticker) *</label>
              <input
                type="text"
                placeholder="e.g. SUN"
                value={symbol}
                onChange={(e) => setSymbol(e.target.value)}
                required
                className="w-full mt-1 bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-xs focus:outline-none focus:border-amber-400"
              />
            </div>
          </div>
        </div>

        {/* Description */}
        <div>
          <label className="text-[10px] font-medium text-slate-400">Token Description *</label>
          <textarea
            rows={3}
            placeholder="Describe your meme token project..."
            value={description}
            onChange={(e) => setDescription(e.target.value)}
            required
            className="w-full mt-1 bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-xs focus:outline-none focus:border-amber-400 resize-none"
          />
        </div>

        {/* Social Links Inputs */}
        <div className="p-3 bg-slate-950 border border-slate-800 rounded-xl space-y-2">
          <span className="text-[11px] font-semibold text-slate-300">Socials (Optional)</span>
          <input
            type="text"
            placeholder="Website URL"
            value={website}
            onChange={(e) => setWebsite(e.target.value)}
            className="w-full bg-slate-900 border border-slate-800 rounded-lg p-2 text-xs focus:outline-none focus:border-amber-400"
          />
          <input
            type="text"
            placeholder="Telegram Group"
            value={telegram}
            onChange={(e) => setTelegram(e.target.value)}
            className="w-full bg-slate-900 border border-slate-800 rounded-lg p-2 text-xs focus:outline-none focus:border-amber-400"
          />
          <input
            type="text"
            placeholder="Twitter / X Profile"
            value={twitter}
            onChange={(e) => setTwitter(e.target.value)}
            className="w-full bg-slate-900 border border-slate-800 rounded-lg p-2 text-xs focus:outline-none focus:border-amber-400"
          />
        </div>

        {/* Initial Buy / Added ICP */}
        <div>
          <label className="text-[10px] font-medium text-slate-400">Initial Buy / Added Liquidity (Optional ICP)</label>
          <input
            type="number"
            min="0"
            placeholder="0.0"
            value={initialBuyIcp}
            onChange={(e) => setInitialBuyIcp(e.target.value)}
            className="w-full mt-1 bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-xs focus:outline-none focus:border-amber-400"
          />
        </div>

        {/* Dynamic Cost Output */}
        <div className="p-3 bg-slate-950/80 rounded-xl border border-slate-800/80 text-xs space-y-1">
          <div className="flex justify-between text-slate-400">
            <span>Treasury Account:</span>

          <div className="flex justify-between text-slate-400">
            <span>Creation Fee:</span>
            <span>{BASE_LAUNCH_FEE_ICP} ICP</span>
          </div>
          {initialBuyNum > 0 && (
            <div className="flex justify-between text-slate-400">
              <span>Initial Buy Amount:</span>
              <span>+{initialBuyNum} ICP</span>
            </div>
          )}
          <div className="flex justify-between text-white font-bold pt-2 border-t border-slate-800">
            <span>Total Payment:</span>
            <span className="text-amber-400 font-mono">{totalDeductionIcp} ICP</span>
          </div>
        </div>

        <button
          type="submit"
          disabled={loading}
          className="w-full bg-amber-400 hover:bg-amber-300 text-slate-950 font-bold py-3.5 rounded-xl text-xs transition-all disabled:opacity-50"
        >
          {loading ? "Deploying Token..." : `Pay ${totalDeductionIcp} ICP & Create Token`}
        </button>
      </form>
    </div>
  );
}

// Dummy CDN Helper
```

### Flow Breakdown

1. **No External Wallets Needed:** Uses your internal e-wallet system (`window.walletActor`) to transfer ICP directly to `USERNAME_TREASURY`.

3. **Dynamic Calculations:** Adding an `Initial Buy` automatically adds to the `5 ICP` base fee, authorizing the total in one transaction.

### Official Motoko Factory Return Types & Fields

The factory returns an asynchronous result variant `#Ok` containing the deployed canister details or `#Err` with an error message.

```motoko
// Official Return Type Variant
public type DeployResult = {
  #Ok : {
    canisterId : Principal;   // Deployed ICRC-1 Token Canister ID
    tokenUrl : Text;         // Full IC HTTP Gateway URL to access token state
    txBlockHeight : Nat;     // Block height proof on ICP Ledger
    createdAt : Int;         // Unix nanosecond timestamp
  };
  #Err : Text;               // Error description if verification or deployment fails
};

```

### Returned Fields Breakdown

| Field Name | Motoko Type | Data & Purpose | Example Value |
| --- | --- | --- | --- |
| `canisterId` | `Principal` | Unique on-chain canister ID generated by the IC Management Canister. | `ryjl3-tyaaa-aaaaa-aaaba-cai` |
| `tokenUrl` | `Text` | Direct HTTPS gateway link serving token metadata & candid interface. | `[https://ryjl3-tyaaa-aaaaa-aaaba-cai.raw.icp0.io](https://ryjl3-tyaaa-aaaaa-aaaba-cai.raw.icp0.io)` |
| `txBlockHeight` | `Nat` | The ICP Ledger block index verifying the **5 ICP** treasury transfer. | `14892031` |
| `createdAt` | `Int` | System time timestamp recorded at the moment of creation. | `1722913800000000000` |

### Standard Motoko Factory Implementation (`src/api/v1/factory.mo`)

```motoko
import Text "mo:base/Text";
import Nat "mo:base/Nat";
import Time "mo:base/Time";
import Principal "mo:base/Principal";

module TokenFactory {

  public type DeployParams = {
    name : Text;
    symbol : Text;
    description : Text;
    imageUrl : Text;
    website : Text;
    telegram : Text;
    twitter : Text;
    initialBuyE8s : Nat;
    txBlockHeight : Nat;
  };

  public type DeployResponse = {
    canisterId : Principal;
    tokenUrl : Text;
    txBlockHeight : Nat;
    createdAt : Int;
  };

  public type DeployResult = {
    #Ok : DeployResponse;
    #Err : Text;
  };

  public class Service() {

    public let USERNAME_TREASURY : Text = "ni5n2-efxui-dyqdu-2mnpr-atclq-d6snc-zdq5q-u6ibz-ibpkq-brjpj-gqe";

    public shared(msg) func deployToken(params : DeployParams) : async DeployResult {
      // 1. Verify ledger payment at params.txBlockHeight
      // 2. Spawn child ICRC-1 ledger canister
      let newCanisterId = Principal.fromText("ryjl3-tyaaa-aaaaa-aaaba-cai"); // Spawned Principal
      let gatewayUrl = "https://" # Principal.toText(newCanisterId) # ".raw.icp0.io";

      return #Ok({
        canisterId = newCanisterId;
        tokenUrl = gatewayUrl;
        txBlockHeight = params.txBlockHeight;
        createdAt = Time.now();
      });
    };
  };
};
```
